-- Create helper function to get translated text
CREATE OR REPLACE FUNCTION get_translation(
    text_ar TEXT,
    text_he TEXT,
    text_en TEXT,
    lang VARCHAR DEFAULT 'ar'
) RETURNS TEXT AS $$
BEGIN
    CASE lang
        WHEN 'he' THEN
            RETURN COALESCE(text_he, text_ar, text_en, '');
        WHEN 'en' THEN
            RETURN COALESCE(text_en, text_ar, text_he, '');
        ELSE
            RETURN COALESCE(text_ar, text_he, text_en, '');
    END CASE;
END;
$$ LANGUAGE plpgsql;

-- Create function to check for suspicious patterns
CREATE OR REPLACE FUNCTION check_suspicious_patterns(p_ip VARCHAR, p_email VARCHAR)
RETURNS TABLE(is_suspicious BOOLEAN, reason TEXT) AS $$
DECLARE
    failed_attempts INT;
BEGIN
    -- Check for multiple failed attempts from same IP (15 minutes)
    SELECT COUNT(*) INTO failed_attempts
    FROM security_logs
    WHERE ip = p_ip
      AND event_type = 'auth_attempt'
      AND success = false
      AND created_at > NOW() - INTERVAL '15 minutes';
    
    IF failed_attempts > 5 THEN
        RETURN QUERY SELECT true, 'Multiple failed attempts from IP ' || p_ip || ' (' || failed_attempts || ' attempts)';
        RETURN;
    END IF;
    
    -- Check for multiple failed attempts for same email (1 hour)
    SELECT COUNT(*) INTO failed_attempts
    FROM security_logs
    WHERE email = p_email
      AND event_type = 'auth_attempt'
      AND success = false
      AND created_at > NOW() - INTERVAL '1 hour';
    
    IF failed_attempts > 10 THEN
        RETURN QUERY SELECT true, 'Multiple failed attempts for email ' || p_email || ' (' || failed_attempts || ' attempts)';
        RETURN;
    END IF;
    
    -- No suspicious patterns detected
    RETURN QUERY SELECT false, 'No suspicious patterns';
END;
$$ LANGUAGE plpgsql;

    p_user_agent TEXT,
    p_success BOOLEAN,

    ) VALUES (
        p_event_type, p_user_id, p_email, p_phone, p_ip, p_user_agent, p_success, p_reason, p_details
    );
END;
$$ LANGUAGE plpgsql;

    text_ar TEXT,
    text_he TEXT,
    text_en TEXT,
    lang VARCHAR DEFAULT 'ar'
) RETURNS TEXT AS $$
BEGIN
    CASE lang
        WHEN 'he' THEN
            RETURN COALESCE(text_he, text_ar, text_en, '');
        WHEN 'en' THEN
            RETURN COALESCE(text_en, text_ar, text_he, '');
        ELSE
            RETURN COALESCE(text_ar, text_he, text_en, '');
    END CASE;
END;
